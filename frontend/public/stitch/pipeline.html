def predict_ensemble(x, trees):
    # Random Forest Aggregation Formula
    predictions = [tree.predict(x) for tree in trees]
    
    return Counter(predictions).most_common(1)[0][0]

# Output represents majority vote:
# ŷ = argmax_c (1/B) Σ I(T_b(x) == c)
